NexaDataGridView

A modernized data grid that wraps System.Windows.Forms.DataGridView with NexaUI theming, command columns, and built-in status/summary/search panels.

How to Add NexaDataGridView Manually

NexaDataGridView inherits from System.Windows.Forms.DataGridView, so all native data-binding, column, and selection behavior is preserved. You can add it via the Designer or in code.

Designer workflow

  1. Build the solution once to load NexaUI controls into the Toolbox.
  2. Drag NexaDataGridView from the Toolbox onto your form.
  3. Set Dock, AutoGenerateColumns, ReadOnly, and other properties in the Properties window.
  4. Add columns in the Designer column editor or in code.
  5. Wire events like SelectionChanged, RowDoubleClicked, or CommandClicked in code-behind.

Creating NexaDataGridView in code

C#using NexaUI.Controls;
using NexaUI.Themes;
using System.Windows.Forms;

public partial class StudentsForm : Form
{
    public StudentsForm()
    {
        InitializeComponent();
        ThemeManager.SetTheme(new LightTheme());
        BuildGrid();
    }

    private void BuildGrid()
    {
        var grid = new NexaDataGridView
        {
            Dock = DockStyle.Fill,
            ReadOnly = true,
            AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill,
            SelectionMode = DataGridViewSelectionMode.FullRowSelect,
            MultiSelect = true,
            EnableSummaryPanel = true,
            EnableSearchPanel = true
        };

        grid.Columns.Add("Id", "Student ID");
        grid.Columns.Add("Name", "Student Name");
        grid.Columns.Add("Course", "Course");
        grid.Columns.Add("Status", "Status");

        var students = GenerateStudents();
        foreach (var s in students)
        {
            grid.Rows.Add(s.Id, s.Name, s.Course, s.Status);
        }

        grid.SelectionChanged += (s, e) =>
        {
            if (grid.SelectedRows.Count > 0)
            {
                var row = grid.SelectedRows[0];
                Console.WriteLine($"Selected: {row.Cells[1].Value}");
            }
        };

        Controls.Add(grid);
    }
}
VB.NETImports NexaUI.Controls
Imports NexaUI.Themes

Public Class StudentsForm
    Public Sub New()
        InitializeComponent()
        ThemeManager.SetTheme(New LightTheme())
        BuildGrid()
    End Sub

    Private Sub BuildGrid()
        Dim grid = New NexaDataGridView With {
            .Dock = DockStyle.Fill,
            .ReadOnly = True,
            .AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill,
            .SelectionMode = DataGridViewSelectionMode.FullRowSelect,
            .MultiSelect = True,
            .EnableSummaryPanel = True,
            .EnableSearchPanel = True
        }

        grid.Columns.Add("Id", "Student ID")
        grid.Columns.Add("Name", "Student Name")
        grid.Columns.Add("Course", "Course")
        grid.Columns.Add("Status", "Status")

        Dim students = GenerateStudents()
        For Each s In students
            grid.Rows.Add(s.Id, s.Name, s.Course, s.Status)
        Next

        AddHandler grid.SelectionChanged, Sub(sender, e)
            If grid.SelectedRows.Count > 0 Then
                Dim row = grid.SelectedRows(0)
                Console.WriteLine($"Selected: {row.Cells(1).Value}")
            End If
        End Sub

        Controls.Add(grid)
    End Sub
End Class

Features

  • Command columns: Built-in Edit and Delete actions with confirmation dialogs
  • Summary panel: Automatic row count and selection totals
  • Search panel: Integrated column-level filtering
  • Theming: Full support for all NexaUI themes
