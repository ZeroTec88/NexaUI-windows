NexaListView

A themed list view that inherits from System.Windows.Forms.ListView with NexaUI styling applied to subitems, groups, and column headers.

How to Add NexaListView Manually

NexaListView inherits from System.Windows.Forms.ListView, so all native views (Details, List, LargeIcon, SmallIcon, Tile), groups, columns, and selection behavior are preserved.

Designer workflow

  1. Build the solution once to load NexaUI controls into the Toolbox.
  2. Drag NexaListView from the Toolbox onto your form.
  3. Set View, Dock, FullRowSelect, and GridLines in the Properties window.
  4. Add columns using the Designer's Columns Editor (right-click → Edit Columns).
  5. Add items in code-behind or at runtime.

Creating NexaListView in code

C#using NexaUI.Controls;
using System.Windows.Forms;

public partial class ReportsForm : Form
{
    public ReportsForm()
    {
        InitializeComponent();
        BuildListView();
    }

    private void BuildListView()
    {
        var list = new NexaListView
        {
            Dock = DockStyle.Fill,
            View = View.Details,
            FullRowSelect = true,
            GridLines = true,
            ListStyle = NexaListStyle.Default
        };

        list.Columns.Add("Name", 220);
        list.Columns.Add("Type", 120);
        list.Columns.Add("Status", 100);

        list.Items.Add(new ListViewItem(new[] { "Student List", "Data", "Ready" }));
        list.Items.Add(new ListViewItem(new[] { "Course List", "Data", "Ready" }));
        list.Items.Add(new ListViewItem(new[] { "Reports", "Document", "Ready" }));

        list.SelectedIndexChanged += (s, e) =>
        {
            if (list.SelectedItems.Count > 0)
            {
                Console.WriteLine($"Selected: {list.SelectedItems[0].Text}");
            }
        };

        Controls.Add(list);
    }
}
VB.NETImports NexaUI.Controls

Public Class ReportsForm
    Public Sub New()
        InitializeComponent()
        BuildListView()
    End Sub

    Private Sub BuildListView()
        Dim list = New NexaListView With {
            .Dock = DockStyle.Fill,
            .View = View.Details,
            .FullRowSelect = True,
            .GridLines = True,
            .ListStyle = NexaListStyle.Default
        }

        list.Columns.Add("Name", 220)
        list.Columns.Add("Type", 120)
        list.Columns.Add("Status", 100)

        list.Items.Add(New ListViewItem(New String() {"Student List", "Data", "Ready"}))
        list.Items.Add(New ListViewItem(New String() {"Course List", "Data", "Ready"}))
        list.Items.Add(New ListViewItem(New String() {"Reports", "Document", "Ready"}))

        AddHandler list.SelectedIndexChanged, Sub(s, e)
            If list.SelectedItems.Count > 0 Then
                Console.WriteLine($"Selected: {list.SelectedItems(0).Text}")
            End If
        End Sub

        Controls.Add(list)
    End Sub
End Class
