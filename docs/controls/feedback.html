Feedback Controls

NexaUI provides visual feedback controls for communicating status, progress, and notifications to users.

NexaProgressBar

A themed progress bar that inherits from System.Windows.Forms.ProgressBar with modern styling.

C#var progress = new NexaProgressBar
{
    Minimum = 0,
    Maximum = 100,
    Value = 45,
    Style = NexaProgressBarStyle.Continuous,
    ShowPercentage = true,
    Width = 300
};
VB.NETDim progress = New NexaProgressBar With {
    .Minimum = 0,
    .Maximum = 100,
    .Value = 45,
    .Style = NexaProgressBarStyle.Continuous,
    .ShowPercentage = True,
    .Width = 300
}

NexaBadge

A compact status badge for labels, counts, and indicators.

C#var badge = new NexaBadge
{
    Text = "New",
    BadgeStyle = NexaBadgeStyle.Success,
    BadgeSize = NexaBadgeSize.Small
};
VB.NETDim badge = New NexaBadge With {
    .Text = "New",
    .BadgeStyle = NexaBadgeStyle.Success,
    .BadgeSize = NexaBadgeSize.Small
}

NexaAlert

An inline notification panel for success, warning, error, and info messages.

C#var alert = new NexaAlert
{
    AlertType = NexaAlertType.Success,
    Title = "Saved",
    Message = "Your changes have been saved.",
    Dismissible = true,
    Width = 400
};
VB.NETDim alert = New NexaAlert With {
    .AlertType = NexaAlertType.Success,
    .Title = "Saved",
    .Message = "Your changes have been saved.",
    .Dismissible = True,
    .Width = 400
}
