Input Controls

NexaUI provides a comprehensive set of input controls that preserve native WinForms behavior while adding modern styling.

How to Add Input Controls Manually

You can add NexaUI input controls using the Designer or by creating them in code. Both approaches work the same way.

Using the Designer

  1. Build the solution once so NexaUI controls appear in the Toolbox.
  2. Open a Form in the Designer.
  3. Drag NexaTextBox, NexaCheckBox, NexaComboBox, etc. from the Toolbox onto the form.
  4. Set properties like Text, Style, Width, and Dock in the Properties window.
  5. Double-click the control to generate an event handler in code-behind.

Creating Input Controls in Code

For dynamic forms, create controls programmatically, set their properties, subscribe to events, and add them to the form's Controls collection.

Complete login form example
C#using NexaUI.Controls;
using NexaUI.Themes;

public partial class LoginForm : Form
{
    public LoginForm()
    {
        InitializeComponent();
        ThemeManager.SetTheme(new LightTheme());
        BuildLoginForm();
    }

    private void BuildLoginForm()
    {
        var card = new NexaCard
        {
            Title = "Sign In",
            Subtitle = "Enter your credentials",
            Width = 420,
            Height = 360,
            CornerRadius = 8,
            ShadowEnabled = true
        };

        var username = new NexaTextBox
        {
            Label = "Username",
            PlaceholderText = "Enter username",
            Icon = NexaIconKind.User,
            Width = 360,
            Dock = DockStyle.Top
        };

        var password = new NexaTextBox
        {
            Label = "Password",
            PlaceholderText = "Enter password",
            UseSystemPasswordChar = true,
            Width = 360,
            Dock = DockStyle.Top
        };

        var remember = new NexaCheckBox
        {
            Text = "Remember me",
            Checked = true,
            Dock = DockStyle.Top
        };

        var loginBtn = new NexaButton
        {
            Text = "Sign In",
            Style = NexaButtonStyle.Primary,
            Width = 120,
            Height = 40,
            Dock = DockStyle.Top
        };
        loginBtn.Click += (s, e) => AttemptLogin(username.Text, password.Text);

        card.ContentPanel.Controls.Add(loginBtn);
        card.ContentPanel.Controls.Add(remember);
        card.ContentPanel.Controls.Add(password);
        card.ContentPanel.Controls.Add(username);

        Controls.Add(card);
    }
}
VB.NETImports NexaUI.Controls
Imports NexaUI.Themes

Public Class LoginForm
    Public Sub New()
        InitializeComponent()
        ThemeManager.SetTheme(New LightTheme())
        BuildLoginForm()
    End Sub

    Private Sub BuildLoginForm()
        Dim card = New NexaCard With {
            .Title = "Sign In",
            .Subtitle = "Enter your credentials",
            .Width = 420,
            .Height = 360,
            .CornerRadius = 8,
            .ShadowEnabled = True
        }

        Dim username = New NexaTextBox With {
            .Label = "Username",
            .PlaceholderText = "Enter username",
            .Icon = NexaIconKind.User,
            .Width = 360,
            .Dock = DockStyle.Top
        }

        Dim password = New NexaTextBox With {
            .Label = "Password",
            .PlaceholderText = "Enter password",
            .UseSystemPasswordChar = True,
            .Width = 360,
            .Dock = DockStyle.Top
        }

        Dim remember = New NexaCheckBox With {
            .Text = "Remember me",
            .Checked = True,
            .Dock = DockStyle.Top
        }

        Dim loginBtn = New NexaButton With {
            .Text = "Sign In",
            .Style = NexaButtonStyle.Primary,
            .Width = 120,
            .Height = 40,
            .Dock = DockStyle.Top
        }
        AddHandler loginBtn.Click, Sub(s, e) AttemptLogin(username.Text, password.Text)

        card.ContentPanel.Controls.Add(loginBtn)
        card.ContentPanel.Controls.Add(remember)
        card.ContentPanel.Controls.Add(password)
        card.ContentPanel.Controls.Add(username)

        Controls.Add(card)
    End Sub
End Class

NexaTextBox

A Bootstrap-style themed text input that hosts a native System.Windows.Forms.TextBox. All native behavior is preserved.

C#var name = new NexaTextBox
{
    Label = "Full name",
    PlaceholderText = "Your full name",
    HelperText = "First and last name as it appears on your ID.",
    Style = NexaInputStyle.Outline,
    InputSize = NexaInputSize.Medium,
    ShowClearButton = true,
    Width = 320
};
VB.NETDim name = New NexaTextBox With {
    .Label = "Full name",
    .PlaceholderText = "Your full name",
    .HelperText = "First and last name as it appears on your ID.",
    .Style = NexaInputStyle.Outline,
    .InputSize = NexaInputSize.Medium,
    .ShowClearButton = True,
    .Width = 320
}

NexaComboBox

VB.NETDim course = New NexaComboBox With {

    .PlaceholderText = "Select a course...",
    .Width = 280
}
course.Items.AddRange(New Object() {"Mathematics", "Physics", "Chemistry"})

    SearchDelayMs = 250,
    Width = 360
};
search.SearchChanged += (s, e) => RunSearch(search.SearchText);
